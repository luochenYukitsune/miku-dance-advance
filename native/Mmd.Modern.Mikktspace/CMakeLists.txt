cmake_minimum_required(VERSION 3.20)
project(MmdModernMikktspace LANGUAGES C)
add_library(mmd_mikktspace SHARED mmd_mikktspace.c upstream/mikktspace.c)
set_target_properties(mmd_mikktspace PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
if(WIN32)
  set_target_properties(mmd_mikktspace PROPERTIES PREFIX "")
endif()
if(MSVC)
  # Static CRT like build.ps1 /MT: the app root ships no VCRUNTIME140.dll for this bridge.
  set_target_properties(mmd_mikktspace PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded")
  target_compile_options(mmd_mikktspace PRIVATE /W4 /fp:precise)
else()
  target_compile_options(mmd_mikktspace PRIVATE -Wall -Wextra -fno-fast-math)
  target_link_libraries(mmd_mikktspace PRIVATE m)
endif()
install(TARGETS mmd_mikktspace RUNTIME DESTINATION . LIBRARY DESTINATION .)
install(FILES LICENSE.MikkTSpace.txt UPSTREAM.json DESTINATION .)
